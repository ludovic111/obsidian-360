import os,pathlib,subprocess,json,hashlib,datetime,platform
base=pathlib.Path('/var/tmp/obsidian-clock-preflight-20261002'); release='6.18.11-xenon-obsidian-clock'
assert platform.release()=='6.18.11-xenon-obsidian4'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
manifest=json.loads((base/'bundle.json').read_text()); expected={m['path']:m['sha256'] for m in manifest['modules']}
extract=base/'inspect';extract.mkdir()

mods=[]
for p in sorted((extract/'usr/lib/modules'/release/'kernel').rglob('*.ko')):

 mods.append({'path':rel,'sha256':sha(p)})
assert len(mods)==43
bins=[]
for name in ['usr/bin/busybox','usr/bin/udevadm','usr/bin/mount','usr/bin/switch_root','usr/bin/e2fsck','usr/bin/kmod']:

 bins.append({'path':name,'sha256':sha(p),'format':'ELF32 big-endian PowerPC'})
assert (extract/'init').is_file() and (extract/'hooks/udev').is_file()
usb=pathlib.Path('/run/obsidian-clock-inspect-usb');usb.mkdir()
assert subprocess.check_output(['blkid','-s','LABEL','-o','value','/dev/sdb1'],text=True).strip()=='ARCH'
assert subprocess.check_output(['blkid','-s','TYPE','-o','value','/dev/sdb1'],text=True).strip()=='vfat'
assert not subprocess.run(['findmnt','-rn','-S','/dev/sdb1'],capture_output=True,text=True).stdout.strip()
files=['kboot.conf','obsidian4/kernel','obsidian4/initrd','linux-hdd/vmlinuz-linux-xenon','linux-hdd/initramfs-linux-xenon.img','arch/boot/ppc/vmlinuz-linux-xenon','arch/boot/ppc/initramfs-linux-xenon.img','Recovery/kboot-before-obsidian4.conf','Recovery/kboot-live-before-install.conf']

try:

 menu=(usb/'kboot.conf').read_text(); assert 'default=linux_hdd\n' in menu and 'obsidian_clock=' not in menu
 # Derive the exact known-good root parameter, avoiding new disk identifiers.

finally:
 subprocess.run(['umount',str(usb)],check=True);usb.rmdir()
report={'console_observed_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'current_release':platform.release(),'modules_verified':mods,'binaries':bins,'init_sha256':sha(extract/'init'),'udev_hook_sha256':sha(extract/'hooks/udev'),'usb_baseline':baseline,'previous_default':'linux_hdd','new_entry':'obsidian_clock','entry_bytes':len(line.encode()),'new_menu_sha256':sha(base/'new-kboot.conf'),'image_sha256':sha(base/'initramfs.img'),'success':True}
(base/'inspection.json').write_text(json.dumps(report,indent=2)+'\n');print(json.dumps(report,indent=2))
